cmake_minimum_required(VERSION 3.16)
project(isaac_manipulation_rviz)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(isaac_manipulation_pointcloud_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(selectable_image SHARED src/selectable_image.cpp)
target_compile_features(selectable_image PUBLIC cxx_std_17)
ament_target_dependencies(selectable_image rclcpp rviz_common pluginlib sensor_msgs isaac_manipulation_pointcloud_interfaces)
target_link_libraries(selectable_image Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS selectable_image LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_image_selection test/test_image_selection.cpp)
  target_include_directories(test_image_selection PRIVATE src)
  ament_target_dependencies(test_image_selection sensor_msgs)
  target_link_libraries(test_image_selection Qt5::Widgets)
endif()
ament_package()
